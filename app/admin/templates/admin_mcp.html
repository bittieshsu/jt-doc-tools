{{ tr('寫進 `opencode.json`，並設定環境變數 `JTDT_TOKEN`：')|marks }}

{
  "$schema": "https://opencode.ai/config.json",
  "mcp": {
    "jt-doc-tools": {
      "type": "remote",
      "url": "{{ url }}",
      "enabled": true,
      "oauth": false,
      "headers": { "Authorization": "Bearer {env:JTDT_TOKEN}" }
    }
  }
}

      "command": "npx",
      "args": ["-y", "mcp-remote", "{{ url }}", "--header", "Authorization:${JTDT_AUTH}"],
      "env": { "JTDT_AUTH": "Bearer <API_TOKEN>" }

  }
